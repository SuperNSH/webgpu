    """Parse a workflow and check what actionlint would catch most often here."""
    import re, sys, yaml
    path = sys.argv[1]
    doc = yaml.safe_load(open(path))
    on = doc.get(True, doc.get("on"))  # PyYAML reads the key `on` as True
    print(f"parsed {path}: name={doc['name']!r}, triggers={sorted(on)}")
    problems = 0
    for name, job in doc["jobs"].items():
        steps = job["steps"]
        assert "runs-on" in job, name
        if "env" in job and "runner." in str(job["env"]):
            print(f"  {name}: runner context in job env"); problems += 1
        for step in steps:
            if "uses" in step:
                ref = step["uses"].split("@", 1)[1]
                if not re.fullmatch(r"[0-9a-f]{40}", ref):
                    print(f"  {name}: unpinned {step['uses']}"); problems += 1
            if not ("uses" in step or "run" in step):
                print(f"  {name}: step without uses/run"); problems += 1
        print(f"  job {name}: runs-on={job['runs-on']}, {len(steps)} steps")
    uses = sorted({s["uses"] for j in doc["jobs"].values() for s in j["steps"] if "uses" in s})
    print("actions:"); [print("  " + u) for u in uses]
    print(f"problems: {problems}")
    sys.exit(1 if problems else 0)
